Нажми «Новый заказ».", mainKeyboard()); }); bot.command("id", async (ctx) => { await ctx.reply(`Ваш user_id: ${ctx.from?.id}`, mainKeyboard()); }); bot.hears(BTN_NEW, async (ctx) => { if (await deny(ctx)) return; const st = getState(ctx); st.step = "entering_order"; st.orderNo = ""; await ctx.reply("Введите номер заказа (например: GF-254):", mainKeyboard()); }); bot.on("text", async (ctx) => { if (await deny(ctx)) return; const st = getState(ctx); const txt = (ctx.message.text || "").trim(); if (txt === BTN_NEW) return; if (st.step === "entering_order") { st.orderNo = txt; st.step = "entering_time"; await ctx.reply("Введите время (минуты, 1–240), например 20:", mainKeyboard()); return; } if (st.step === "entering_time") { const n = Number(txt); if (!Number.isFinite(n) || n < 1 || n > 240) { await ctx.reply("Введите число 1–240.", mainKeyboard()); return; } // ✅ Here: as soon as manager enters time -> order flies to your screen addOrder(st.orderNo, Math.floor(n));
